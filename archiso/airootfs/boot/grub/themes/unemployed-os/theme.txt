# Unemployed OS GRUB theme -- the OEM-style boot logo for the *installed*
# system (the syslinux splash under archiso/syslinux/splash.png covers the
# live medium's own BIOS boot menu; this is the separate, second bootloader
# moment on every real boot after install).
#
# Ships at /boot/grub/themes/unemployed-os/ via the airootfs overlay so
# Calamares' unpackfs (which clones the live squashfs onto the install
# target verbatim) carries it over automatically -- same trick already
# used for Calamares' own branding files, see customize_airootfs.sh.
# Applied via GRUB_THEME in /etc/default/grub (also in this overlay).
#
# Font: "GNU Unifont Regular 16" is the name embedded in unicode.pf2,
# which ships with the grub package itself (already in packages.x86_64)
# at /usr/share/grub/unicode.pf2 -- grub-install copies it into
# /boot/grub/fonts/unicode.pf2 on install, and grub-mkconfig's generated
# grub.cfg already `loadfont`s it. No custom font needed.

title-text: ""
desktop-image: "background.png"
desktop-color: "#08090b"
terminal-font: "GNU Unifont Regular 16"

}

}
